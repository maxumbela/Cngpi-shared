                      {soc.handle}
                    </p>
                  </div>

                  <p className="text-xs text-slate-400">
                    {lang === 'bn' ? soc.descBn : soc.descEn}
                  </p>
                </div>

                <div className="pt-4 mt-4 border-t border-white/10 flex items-center justify-between gap-2">
                  <button
                    onClick={() => handleCopy(soc.handle, soc.id)}
                    className="px-2.5 py-1.5 rounded-lg bg-white/5 hover:bg-white/10 text-xs text-slate-300 flex items-center gap-1.5 transition cursor-pointer"
                    title="Copy Handle"
                  >
                    {copiedId === soc.id ? (
                      <Check className="w-3.5 h-3.5 text-emerald-400" />
                    ) : (
                      <Copy className="w-3.5 h-3.5 text-cyan-400" />
                    )}
                    <span>{copiedId === soc.id ? 'Copied' : 'Copy'}</span>
                  </button>

                  <a
                    href={soc.url}
                    target="_blank"
                    rel="noopener noreferrer"
                    className="px-3 py-1.5 rounded-lg bg-cyan-500/15 hover:bg-cyan-400 hover:text-slate-950 text-cyan-300 text-xs font-semibold flex items-center gap-1 transition"
                  >
                    <span>{lang === 'bn' ? 'ভিজিট' : 'Visit'}</span>
                    <ExternalLink className="w-3.5 h-3.5" />
                  </a>
                </div>
              </div>
            ))}
          </div>
        </section>

        {/* ==========================================
            5. EXPERIENCE & EDUCATION TIMELINE
           ========================================== */}
        <section id="journey" className="space-y-6">
          <div>
            <div className="inline-flex items-center gap-2 text-xs font-mono uppercase tracking-widest text-emerald-400 mb-2">
              <Compass className="w-4 h-4" />
              <span>{lang === 'bn' ? 'পেশাগত অভিজ্ঞতা ও শিক্ষা' : 'Career & Academic Path'}</span>
            </div>
            <h2 className="text-2xl sm:text-4xl font-bold text-white">
              {lang === 'bn' ? 'আমার যাত্রা ও শিক্ষাগত যোগ্যতা' : 'Experience & Education Timeline'}
            </h2>
          </div>

          <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
            {data.timeline.map((item) => (
              <div
                key={item.id}
                className="rounded-2xl bg-white/[0.03] border border-white/10 p-6 relative overflow-hidden flex flex-col justify-between"
              >
                <div className="space-y-3">
                  <div className="flex items-center justify-between">
                    <span className="px-3 py-1 rounded-full text-xs font-mono bg-cyan-500/10 text-cyan-300 border border-cyan-500/20">
                      {lang === 'bn' ? item.periodBn : item.periodEn}
                    </span>
                    {item.type === 'work' && <Briefcase className="w-5 h-5 text-amber-400" />}
                    {item.type === 'creator' && <Sparkles className="w-5 h-5 text-fuchsia-400" />}
                    {item.type === 'education' && <GraduationCap className="w-5 h-5 text-emerald-400" />}
                  </div>

                  <h3 className="text-xl font-bold text-white">
                    {lang === 'bn' ? item.roleBn : item.roleEn}
                  </h3>

                  <p className="text-xs sm:text-sm font-semibold text-cyan-300">
                    {lang === 'bn' ? item.orgBn : item.orgEn}
                  </p>

                  <p className="text-sm text-slate-300 leading-relaxed">
                    {lang === 'bn' ? item.descBn : item.descEn}
                  </p>
                </div>
              </div>
            ))}
          </div>
        </section>
      </main>

      {/* ==========================================
          FOOTER
         ========================================== */}
      <footer className="relative z-10 border-t border-white/10 bg-[#04050c] py-10">
        <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 flex flex-col sm:flex-row items-center justify-between gap-4 text-xs text-slate-400">
          <div className="flex items-center gap-2">
            <span className="font-bold text-white">
              {lang === 'bn' ? data.profile.nameBn : data.profile.nameEn}
            </span>
            <span>•</span>
            <span>Qalpik (@qalpikai.official)</span>
          </div>
          <div className="flex items-center gap-4">
            <button
              onClick={() => setIsAdminOpen(true)}
              className="text-cyan-400 hover:underline cursor-pointer"
            >
              {lang === 'bn' ? 'অ্যাডমিন প্যানেল ও লিংক এডিটর' : 'Open Admin Customizer'}
            </button>
            <span>•</span>
            <span>Built with Cyber-Glass Architecture</span>
          </div>
        </div>
      </footer>

      {/* ==========================================
          6. LIVE ADMIN PANEL / CUSTOMIZER MODAL
         ========================================== */}
      {isAdminOpen && (
        <div className="fixed inset-0 z-50 bg-black/80 backdrop-blur-md flex items-center justify-center p-4 overflow-y-auto">
          <div className="bg-[#0b0d1e] border border-cyan-500/40 rounded-3xl max-w-2xl w-full p-6 sm:p-8 space-y-6 max-h-[90vh] overflow-y-auto shadow-[0_0_50px_rgba(34,211,238,0.25)]">
            <div className="flex items-center justify-between border-b border-white/10 pb-4">
              <div className="flex items-center gap-2.5">
                <Settings className="w-5 h-5 text-cyan-400" />
                <h3 className="text-lg sm:text-xl font-bold text-white">
                  {lang === 'bn'
                    ? 'ওয়েবসাইট কাস্টমাইজার (Admin Panel)'
                    : 'Live Portfolio Admin & Link Manager'}
                </h3>
              </div>
              <button
                onClick={() => setIsAdminOpen(false)}
                className="p-2 rounded-lg bg-white/5 hover:bg-white/10 text-slate-300 cursor-pointer"
              >
                <X className="w-5 h-5" />
              </button>
            </div>

            {/* Edit Bio Section */}
            <div className="space-y-3">
              <h4 className="text-xs font-mono uppercase tracking-wider text-cyan-400">
                {lang === 'bn' ? '১. আপনার বায়ো ও পরিচয় আপডেট করুন' : '1. Edit Bio & Tagline'}
              </h4>
              <div className="grid grid-cols-1 gap-3">
                <input
                  type="text"
                  value={data.profile.nameBn}
                  onChange={(e) =>
                    setData({
                      ...data,
                      profile: { ...data.profile, nameBn: e.target.value }
                    })
                  }
                  placeholder="নাম (বাংলায়)"
                  className="w-full px-3.5 py-2 rounded-xl bg-slate-900 border border-white/15 text-sm text-white"
                />
                <textarea
                  rows={2}
                  value={data.profile.bioBn}
                  onChange={(e) =>
                    setData({
                      ...data,
                      profile: { ...data.profile, bioBn: e.target.value }
                    })
                  }
                  placeholder="সংক্ষিপ্ত বায়ো (বাংলায়)"
                  className="w-full px-3.5 py-2 rounded-xl bg-slate-900 border border-white/15 text-sm text-white"
                />
              </div>
            </div>

            {/* Add New Project Form */}
            <form onSubmit={handleAddProject} className="space-y-3 border-t border-white/10 pt-4">
              <h4 className="text-xs font-mono uppercase tracking-wider text-fuchsia-400">
                {lang === 'bn' ? '২. নতুন প্রজেক্ট যোগ করুন' : '2. Add a New Project Card'}
              </h4>
              <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
                <input
                  type="text"
                  placeholder="প্রজেক্টের নাম (যেমন: AI Bot)"
                  value={newProject.titleBn}
                  onChange={(e) =>
                    setNewProject({ ...newProject, titleBn: e.target.value, titleEn: e.target.value })
                  }
                  className="px-3.5 py-2 rounded-xl bg-slate-900 border border-white/15 text-sm text-white"
                />
                <select
                  value={newProject.category}
                  onChange={(e) => setNewProject({ ...newProject, category: e.target.value })}
                  className="px-3.5 py-2 rounded-xl bg-slate-900 border border-white/15 text-sm text-white"
                >
                  <option value="web">Web & Software</option>
                  <option value="ai">AI & Video Content</option>
                  <option value="iot">ESP32 & Hardware</option>
                </select>
              </div>
              <input
                type="text"
                placeholder="প্রজেক্টের বিবরণ..."
                value={newProject.descBn}
                onChange={(e) =>
                  setNewProject({ ...newProject, descBn: e.target.value, descEn: e.target.value })
                }
                className="w-full px-3.5 py-2 rounded-xl bg-slate-900 border border-white/15 text-sm text-white"
              />
              <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
                <input
                  type="text"
                  placeholder="টেকনোলজি (কমা দিয়ে লিখুন: React, AI)"
                  value={newProject.tech}
                  onChange={(e) => setNewProject({ ...newProject, tech: e.target.value })}
                  className="px-3.5 py-2 rounded-xl bg-slate-900 border border-white/15 text-sm text-white"
                />
                <input
                  type="text"
                  placeholder="প্রজেক্ট লিংক (https://...)"
                  value={newProject.link}
                  onChange={(e) => setNewProject({ ...newProject, link: e.target.value })}
                  className="px-3.5 py-2 rounded-xl bg-slate-900 border border-white/15 text-sm text-white"
                />
              </div>
              <button
                type="submit"
                className="px-4 py-2 rounded-xl bg-cyan-400 text-slate-950 font-bold text-xs flex items-center gap-1.5 cursor-pointer"
              >
                <Plus className="w-4 h-4" />
                <span>{lang === 'bn' ? 'প্রজেক্ট যুক্ত করুন' : 'Add Project'}</span>
              </button>
            </form>

            {/* Manage Existing Projects */}
            <div className="space-y-2 border-t border-white/10 pt-4">
              <h4 className="text-xs font-mono uppercase tracking-wider text-slate-400">
                {lang === 'bn' ? 'বর্তমান প্রজেক্ট তালিকা:' : 'Current Projects:'}
              </h4>
              <div className="space-y-2 max-h-40 overflow-y-auto pr-1">
                {data.projects.map((p) => (
                  <div
                    key={p.id}
                    className="flex items-center justify-between px-3 py-2 rounded-lg bg-white/5 text-xs"
                  >
                    <span className="font-medium text-white">{p.titleBn}</span>
                    <button
                      onClick={() => handleDeleteProject(p.id)}
                      className="text-rose-400 hover:text-rose-300 flex items-center gap-1 cursor-pointer"
                    >
                      <Trash2 className="w-3.5 h-3.5" />
                    </button>
                  </div>
                ))}
              </div>
            </div>

            {/* Footer Controls */}
            <div className="flex items-center justify-between border-t border-white/10 pt-4">
              <button
                onClick={handleResetDefault}
                className="px-3.5 py-2 rounded-xl bg-rose-500/15 hover:bg-rose-500/25 text-rose-300 text-xs font-semibold flex items-center gap-1.5 cursor-pointer"
              >
                <RotateCcw className="w-3.5 h-3.5" />
                <span>{lang === 'bn' ? 'ডিফল্ট রিসেট করুন' : 'Reset Defaults'}</span>
              </button>

              <button
                onClick={() => setIsAdminOpen(false)}
                className="px-5 py-2 rounded-xl bg-gradient-to-r from-cyan-400 to-fuchsia-500 text-slate-950 font-bold text-xs cursor-pointer"
              >
                {lang === 'bn' ? 'সেভ ও বন্ধ করুন' : 'Save & Close'}
              </button>
            </div>
          </div>
        </div>
      )}
    </div>
  );
}
